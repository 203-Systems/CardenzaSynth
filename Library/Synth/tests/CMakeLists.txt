cmake_minimum_required(VERSION 3.20)
project(SynthRegression LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(CARDENZA_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CARDENZA_ROOT}/cmake/CardenzaSynthSourceLists.cmake")
cardenza_collect_amy_c_sources("${CARDENZA_ROOT}" amy_sources)
cardenza_collect_synth_core_cpp_sources("${CARDENZA_ROOT}" synth_sources)
cardenza_collect_midi_framework_cpp_sources("${CARDENZA_ROOT}" midi_sources)
cardenza_collect_synth_fx_cpp_sources("${CARDENZA_ROOT}" fx_sources)
add_library(RegressionAmy STATIC ${amy_sources})
target_include_directories(RegressionAmy PUBLIC "${CARDENZA_ROOT}/Library/third_party/amy/src" "${CARDENZA_ROOT}")
target_compile_definitions(RegressionAmy PUBLIC AMY_WAVETABLE AMY_NO_MINIAUDIO _CRT_SECURE_NO_WARNINGS)
add_executable(SynthRegression synth_regression.cpp ${synth_sources} ${midi_sources} ${fx_sources})
target_link_libraries(SynthRegression PRIVATE RegressionAmy)
# The fork's own checks of AMY's control arithmetic (silent levels, DX7 envelopes that froze after
# 5.8 s), against the AMY build the synth uses.
add_executable(SynthAmyControlTests "${CARDENZA_ROOT}/Library/third_party/amy/tests/test_dsp_control.c")
target_link_libraries(SynthAmyControlTests PRIVATE RegressionAmy)
if(WIN32)
  # windows.h's min / max macros break std::numeric_limits<...>::max() in the runtime.
  target_compile_definitions(RegressionAmy PUBLIC NOMINMAX)
endif()
if(MSVC)
  target_compile_options(RegressionAmy PRIVATE $<$<COMPILE_LANGUAGE:C>:/std:c11>)
  target_compile_options(SynthRegression PRIVATE /UNDEBUG)
else()
  target_compile_options(SynthRegression PRIVATE -UNDEBUG)
  target_link_libraries(RegressionAmy PUBLIC m pthread)
endif()
# The test build's AMY sources and drum kits (Library/Synth/Audio/Amy/AmyBench.h), checked on the host:
# -DCARDENZA_AMY_BENCH=ON, then CARDENZA_GAMMA9001_BIN=<drums.bin> AmyBenchCheck.
option(CARDENZA_AMY_BENCH "Build the check of the test build's AMY sources" OFF)
if(CARDENZA_AMY_BENCH)
  add_library(RegressionAmyBench STATIC ${amy_sources})
  target_include_directories(RegressionAmyBench PUBLIC "${CARDENZA_ROOT}/Library/third_party/amy/src" "${CARDENZA_ROOT}")
  target_compile_definitions(RegressionAmyBench PUBLIC AMY_WAVETABLE AMY_NO_MINIAUDIO GAMMA9001 CARDENZA_AMY_BENCH=1 _CRT_SECURE_NO_WARNINGS)
  add_executable(AmyBenchCheck amy_bench_check.cpp ${synth_sources} ${midi_sources} ${fx_sources})
  target_link_libraries(AmyBenchCheck PRIVATE RegressionAmyBench)
  if(NOT MSVC)
    target_compile_options(AmyBenchCheck PRIVATE -UNDEBUG)
    target_link_libraries(RegressionAmyBench PUBLIC m pthread)
  endif()
endif()
enable_testing()
add_test(NAME SynthRegression COMMAND SynthRegression)
add_test(NAME SynthAmyControlTests COMMAND SynthAmyControlTests)
set_tests_properties(SynthAmyControlTests PROPERTIES TIMEOUT 15)
# With the Plaits engine's presets and voices rendered too (an unoptimised build renders them slowly).
set_tests_properties(SynthRegression PROPERTIES TIMEOUT 30)
