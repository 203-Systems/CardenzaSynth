cmake_minimum_required(VERSION 3.20)
project(SynthRegression LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(CARDENZA_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${CARDENZA_ROOT}/cmake/CardenzaSynthSourceLists.cmake")
cardenza_collect_amy_c_sources("${CARDENZA_ROOT}" amy_sources)
cardenza_collect_synth_core_cpp_sources("${CARDENZA_ROOT}" synth_sources)
cardenza_collect_midi_framework_cpp_sources("${CARDENZA_ROOT}" midi_sources)
cardenza_collect_synth_fx_cpp_sources("${CARDENZA_ROOT}" fx_sources)
add_library(RegressionAmy STATIC ${amy_sources})
target_include_directories(RegressionAmy PUBLIC "${CARDENZA_ROOT}/Library/third_party/amy/src" "${CARDENZA_ROOT}")
target_compile_definitions(RegressionAmy PUBLIC AMY_WAVETABLE AMY_NO_MINIAUDIO _CRT_SECURE_NO_WARNINGS)
add_executable(SynthRegression synth_regression.cpp ${synth_sources} ${midi_sources} ${fx_sources})
target_link_libraries(SynthRegression PRIVATE RegressionAmy)
if(MSVC)
  target_compile_options(RegressionAmy PRIVATE $<$<COMPILE_LANGUAGE:C>:/std:c11>)
  target_compile_options(SynthRegression PRIVATE /UNDEBUG)
else()
  target_compile_options(SynthRegression PRIVATE -UNDEBUG)
  target_link_libraries(RegressionAmy PUBLIC m pthread)
endif()
enable_testing()
add_test(NAME SynthRegression COMMAND SynthRegression)
set_tests_properties(SynthRegression PROPERTIES TIMEOUT 15)
